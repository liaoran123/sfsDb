## 5.1 完整的可运行示例

让我们通过一个完整的示例来演示 sfsDb 的表组合查询功能：

```go
package main

import (
	"fmt"
	"os"

	"github.com/liaoran123/sfsDb/engine"
	"github.com/liaoran123/sfsDb/match"
	"github.com/liaoran123/sfsDb/storage"
)

func main() {
	fmt.Println("=== sfsDb 数据库实战 - 第 5 章：表组合查询示例 ===")
	fmt.Println()

	dbPath := "./bible_examples_chapter05_db"
	cleanup(dbPath)

	fmt.Println("1. 初始化数据库")
	_, err := storage.GetDBManager().OpenDB(dbPath)
	if err != nil {
		fmt.Printf("数据库打开失败: %v\n", err)
		return
	}
	defer storage.GetDBManager().CloseDB()
	fmt.Println("   ✓ 数据库打开成功")
	fmt.Println()

	fmt.Println("2. 创建表 1 - 用户表")
	table1, err := engine.TableNew("users")
	if err != nil {
		fmt.Printf("创建用户表失败: %v\n", err)
		return
	}

	fields1 := map[string]any{
		"id":     0,
		"name":   "",
		"age":    0,
		"active": false,
	}
	err = table1.SetFields(fields1)
	if err != nil {
		fmt.Printf("设置字段失败: %v\n", err)
		return
	}

	pk1, _ := engine.DefaultPrimaryKeyNew("pk_users")
	pk1.AddFields("id")
	table1.CreateIndex(pk1)
	fmt.Println("   ✓ 用户表创建成功")
	fmt.Println()

	fmt.Println("3. 创建表 2 - 订单表")
	table2, err := engine.TableNew("orders")
	if err != nil {
		fmt.Printf("创建订单表失败: %v\n", err)
		return
	}

	fields2 := map[string]any{
		"order_id": 0,
		"user_id":  0,
		"product":  "",
		"amount":   0.0,
	}
	err = table2.SetFields(fields2)
	if err != nil {
		fmt.Printf("设置字段失败: %v\n", err)
		return
	}

	pk2, _ := engine.DefaultPrimaryKeyNew("pk_orders")
	pk2.AddFields("order_id")
	table2.CreateIndex(pk2)
	fmt.Println("   ✓ 订单表创建成功")
	fmt.Println()

	fmt.Println("4. 插入用户数据")
	users := []map[string]any{
		{"id": 1, "name": "Alice", "age": 20, "active": true},
		{"id": 2, "name": "Bob", "age": 25, "active": true},
		{"id": 3, "name": "Charlie", "age": 30, "active": false},
		{"id": 4, "name": "David", "age": 35, "active": true},
		{"id": 5, "name": "Eve", "age": 40, "active": false},
	}

	for _, u := range users {
		_, err = table1.Insert(&amp;u)
		if err != nil {
			fmt.Printf("插入用户数据失败: %v\n", err)
			return
		}
	}
	fmt.Println("   ✓ 成功插入", len(users), "条用户数据")
	fmt.Println()

	fmt.Println("5. 插入订单数据")
	orders := []map[string]any{
		{"order_id": 101, "user_id": 1, "product": "Go语言入门", "amount": 59.0},
		{"order_id": 102, "user_id": 2, "product": "Go语言进阶", "amount": 79.0},
		{"order_id": 103, "user_id": 3, "product": "机械键盘", "amount": 399.0},
		{"order_id": 104, "user_id": 4, "product": "无线鼠标", "amount": 129.0},
		{"order_id": 105, "user_id": 3, "product": "显示器", "amount": 1999.0},
	}

	for _, o := range orders {
		_, err = table2.Insert(&amp;o)
		if err != nil {
			fmt.Printf("插入订单数据失败: %v\n", err)
			return
		}
	}
	fmt.Println("   ✓ 成功插入", len(orders), "条订单数据")
	fmt.Println()

	fmt.Println("6. 获取两个表的所有记录")
	iter1, _ := table1.Search(&amp;map[string]any{"id": nil})
	defer engine.GlobalTableIterPool.Put(iter1)
	iter2, _ := table2.Search(&amp;map[string]any{"order_id": nil})
	defer engine.GlobalTableIterPool.Put(iter2)

	fmt.Println("--------- 用户表记录 ---------")
	rd1 := iter1.GetRecords(true)
	for _, record := range rd1 {
		fmt.Println(record)
	}
	rd1.Release()

	fmt.Println("--------- 订单表记录 ---------")
	rd2 := iter2.GetRecords(true)
	for _, record := range rd2 {
		fmt.Println(record)
	}
	rd2.Release()
	fmt.Println()

	fmt.Println("7. 表组合查询 1 - IN 查询（有订单的用户）")
	fmt.Println("   SQL 等价: SELECT users.* FROM users, orders WHERE users.id = orders.user_id")
	fmt.Println("-----------------------------------------------")
	map2 := iter2.Map("user_id")
	defer iter2.ReleaseMap(map2)
	mach := match.NewAND([]string{"id"}, map2)
	iter1.SetMatch(mach)
	rd3 := iter1.GetRecords(true)
	for _, record := range rd3 {
		fmt.Println(record)
	}
	rd3.Release()
	fmt.Println()

	fmt.Println("8. 表组合查询 2 - NOT IN 查询（没有订单的用户）")
	fmt.Println("   SQL 等价: SELECT users.* FROM users, orders WHERE users.id != orders.user_id")
	fmt.Println("-----------------------------------------------")
	mach1 := match.NewAND([]string{"id"}, map2, false)
	iter1.SetMatch(mach1)
	rd4 := iter1.GetRecords(true)
	for _, record := range rd4 {
		fmt.Println(record)
	}
	rd4.Release()
	fmt.Println()

	fmt.Println("9. 字段比较查询 - 年龄大于 25 的用户")
	fmt.Println("   SQL 等价: SELECT users.* FROM users WHERE users.age &gt; 25")
	fmt.Println("-----------------------------------------------")
	iter1, _ = table1.Search(&amp;map[string]any{"id": nil})
	defer engine.GlobalTableIterPool.Put(iter1)
	ageMatcher := match.NewGreaterThanMatch("age", 25)
	iter1.SetMatch(ageMatcher)
	rd5 := iter1.GetRecords(true)
	for _, record := range rd5 {
		fmt.Println(record)
	}
	rd5.Release()
	fmt.Println()

	fmt.Println("10. 组合查询 - 年龄大于 25 且活跃的用户")
	fmt.Println("    SQL 等价: SELECT users.* FROM users WHERE users.age &gt; 25 AND users.active = true")
	fmt.Println("-----------------------------------------------")
	iter1, _ = table1.Search(&amp;map[string]any{"id": nil})
	defer engine.GlobalTableIterPool.Put(iter1)
	activeMatcher := match.NewEqualMatch("active", true)
	iter1.SetMatch(ageMatcher, activeMatcher)
	rd6 := iter1.GetRecords(true)
	for _, record := range rd6 {
		fmt.Println(record)
	}
	rd6.Release()
	fmt.Println()

	fmt.Println("=== 表组合查询示例完成！===")
}

func cleanup(path string) {
	os.RemoveAll(path)
}
```

这段代码展示了 sfsDb 的完整表组合查询功能：

1. **初始化数据库**：通过 `storage.GetDBManager().OpenDB()` 打开数据库
2. **创建表**：使用 `engine.TableNew()` 创建多个表
3. **设置字段**：通过 `SetFields()` 定义表结构
4. **插入数据**：使用 `Insert()` 插入数据到多个表
5. **获取迭代器**：使用 `Search()` 获取表的迭代器
6. **Map() 方法**：将迭代器数据转换为 map，用于匹配
7. **NewAND() 匹配器**：创建 IN/NOT IN 匹配器
8. **字段比较匹配器**：使用 `NewGreaterThanMatch()`、`NewEqualMatch()` 等
9. **SetMatch() 方法**：设置匹配条件到迭代器
10. **组合匹配**：同时使用多个匹配器

## 5.2 Match 接口概述

sfsDb 的表组合查询功能通过 `Match` 接口实现，该接口定义在 `match/match.go` 中：

```go
type Match interface {
    Match(fields *map[string]any) bool
}
```

**Match 接口**：根据 TableIter 回传的 key, value 所有能得到的 field 的值，进行所需匹配。

## 5.3 AND 匹配器（IN/NOT IN 查询）

AND 匹配器用于实现类似 SQL 的 IN 和 NOT IN 查询，使用多个迭代器的主键值进行匹配。

### 5.3.1 AND 结构体

```go
type AND struct {
    fields []string        // 需要匹配的字段名
    data   map[any]bool    // 由 Map() 方法生成的数据
    rule   bool            // true 为 IN，false 为 NOT IN
}
```

### 5.3.2 创建 AND 匹配器

```go
func NewAND(fields []string, data map[any]bool, rule ...bool) *AND {
    if len(rule) == 0 {
        rule = append(rule, true)
    }
    return &amp;AND{
        fields: fields,
        data:   data,
        rule:   rule[0],
    }
}
```

**关键参数**：
- `fields`：需要匹配的字段名数组
- `data`：由 `TableIter.Map()` 生成的 map
- `rule`：匹配规则，true 为 IN，false 为 NOT IN（默认 true）

---

## 5.4 字段比较匹配器

字段比较匹配器用于对单个字段进行各种比较操作，支持等于、不等于、大于、小于等操作。

### 5.4.1 比较操作符

```go
type ComparisonOperator int

const (
    Equal              ComparisonOperator = iota  // 等于 (=)
    NotEqual                                       // 不等于 (!=)
    GreaterThan                                    // 大于 (&gt;)
    GreaterThanOrEqual                             // 大于等于 (&gt;=)
    LessThan                                       // 小于 (&lt;)
    LessThanOrEqual                                // 小于等于 (&lt;=)
    Like                                           // 类似于 SQL LIKE 操作
    Prefix                                         // 前缀匹配
    Suffix                                         // 后缀匹配
    Contains                                       // 包含匹配
)
```

### 5.4.2 创建字段比较匹配器

```go
// 创建相等匹配器
func NewEqualMatch(fieldName string, value any) *FieldComparison {
    return NewFieldComparison(fieldName, Equal, value)
}

// 创建大于匹配器
func NewGreaterThanMatch(fieldName string, value any) *FieldComparison {
    return NewFieldComparison(fieldName, GreaterThan, value)
}

// 创建包含匹配器
func NewContainsMatch(fieldName string, substr string) *FieldComparison {
    return NewFieldComparison(fieldName, Contains, substr)
}
```
