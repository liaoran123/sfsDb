## 4.1 完整的可运行示例

让我们通过一个完整的示例来演示 sfsDb 的索引与查询优化功能：

```go
package main

import (
	"fmt"
	"os"

	"github.com/liaoran123/sfsDb/engine"
	"github.com/liaoran123/sfsDb/storage"
)

func main() {
	fmt.Println("=== sfsDb 数据库实战 - 第 4 章：索引与查询优化示例 ===")
	fmt.Println()

	dbPath := "./bible_examples_chapter04_db"
	cleanup(dbPath)

	fmt.Println("1. 初始化数据库")
	_, err := storage.GetDBManager().OpenDB(dbPath)
	if err != nil {
		fmt.Printf("数据库打开失败: %v\n", err)
		return
	}
	defer storage.GetDBManager().CloseDB()
	fmt.Println("   ✓ 数据库打开成功")
	fmt.Println()

	fmt.Println("2. 创建表并设置字段")
	table, err := engine.TableNew("products")
	if err != nil {
		fmt.Printf("创建表失败: %v\n", err)
		return
	}

	fields := map[string]any{
		"id":          0,
		"name":        "",
		"category":    "",
		"price":       0,
		"description": "",
	}
	err = table.SetFields(fields)
	if err != nil {
		fmt.Printf("设置字段失败: %v\n", err)
		return
	}
	fmt.Println("   ✓ 表创建成功")
	fmt.Println()

	fmt.Println("3. 创建各种索引")
	primaryKey, _ := engine.DefaultPrimaryKeyNew("pk_id")
	primaryKey.AddFields("id")
	table.CreateIndex(primaryKey)
	fmt.Println("   ✓ 主键索引创建成功")

	nameIndex, _ := engine.DefaultNormalIndexNew("idx_name")
	nameIndex.AddFields("name")
	table.CreateIndex(nameIndex)
	fmt.Println("   ✓ 商品名称索引创建成功")

	categoryPriceIndex, _ := engine.DefaultNormalIndexNew("idx_category_price")
	categoryPriceIndex.AddFields("category", "price")
	table.CreateIndex(categoryPriceIndex)
	fmt.Println("   ✓ 分类-价格复合索引创建成功")

	fullTextIndex, _ := engine.DefaultFullTextIndexNew("ft_description")
	fullTextIndex.AddFields("description", "id")
	fullTextIndex.SetFullField("description", 5)
	table.CreateIndex(fullTextIndex)
	fmt.Println("   ✓ 全文索引创建成功")
	fmt.Println()

	fmt.Println("4. 插入测试数据")
	products := []map[string]any{
		{
			"id":          1,
			"name":        "Go语言入门",
			"category":    "图书",
			"price":       59,
			"description": "Go语言是一种开源的编程语言",
		},
		{
			"id":          2,
			"name":        "Go语言进阶",
			"category":    "图书",
			"price":       79,
			"description": "深入学习Go语言的并发编程",
		},
		{
			"id":          3,
			"name":        "机械键盘",
			"category":    "电子产品",
			"price":       399,
			"description": "高品质青轴机械键盘",
		},
		{
			"id":          4,
			"name":        "无线鼠标",
			"category":    "电子产品",
			"price":       129,
			"description": "人体工学无线鼠标",
		},
	}

	for _, p := range products {
		_, err = table.Insert(&amp;p)
		if err != nil {
			fmt.Printf("插入数据失败: %v\n", err)
			return
		}
	}
	fmt.Println("   ✓ 成功插入", len(products), "条商品数据")
	fmt.Println()

	fmt.Println("5. 测试主键搜索")
	searchPK := map[string]any{"id": 2}
	dataIter, _ := table.Search(&amp;searchPK)
	defer dataIter.Release()
	records := dataIter.GetRecords(true)
	defer records.Release()
	if len(records) &gt; 0 {
		fmt.Printf("   主键搜索结果: %+v\n", records[0])
	}
	fmt.Println()

	fmt.Println("6. 测试普通索引搜索（按名称）")
	searchName := map[string]any{"name": "机械键盘"}
	dataIter2, _ := table.Search(&amp;searchName)
	defer dataIter2.Release()
	records2 := dataIter2.GetRecords(true)
	defer records2.Release()
	if len(records2) &gt; 0 {
		fmt.Printf("   名称搜索结果: %+v\n", records2[0])
	}
	fmt.Println()

	fmt.Println("7. 测试复合索引搜索（按分类）")
	searchCategory := map[string]any{"category": "图书"}
	dataIter3, _ := table.Search(&amp;searchCategory)
	defer dataIter3.Release()
	records3 := dataIter3.GetRecords(true)
	defer records3.Release()
	fmt.Println("   分类'图书'的商品:")
	for _, p := range records3 {
		fmt.Printf("   - %s: ¥%d\n", p["name"], p["price"])
	}
	fmt.Println()

	fmt.Println("8. 测试全文索引搜索")
	searchFT := map[string]any{"description": "Go语言"}
	dataIter4, _ := table.Search(&amp;searchFT)
	defer dataIter4.Release()
	records4 := dataIter4.GetRecords(true)
	defer records4.Release()
	fmt.Println("   描述包含'Go语言'的商品:")
	for _, p := range records4 {
		fmt.Printf("   - %s\n", p["name"])
	}
	fmt.Println()

	fmt.Println("9. 查看索引缓存统计")
	stats := engine.GetIndexMatchCacheStats()
	fmt.Printf("   索引缓存 - 大小: %d, 访问: %d, 命中: %d, 命中率: %.2f%%\n",
		stats.Size, stats.Accesses, stats.Hits, stats.HitRate*100)
	fmt.Println()

	fmt.Println("=== 索引与查询优化示例完成！===")
}

func cleanup(path string) {
	os.RemoveAll(path)
}
```

这段代码展示了 sfsDb 的完整索引与查询优化功能：

1. **初始化数据库**：通过 `storage.GetDBManager().OpenDB()` 打开数据库
2. **创建表**：使用 `engine.TableNew()` 创建表
3. **设置字段**：通过 `SetFields()` 定义表结构
4. **创建主键索引**：使用 `DefaultPrimaryKeyNew()` 创建主键索引
5. **创建普通索引**：使用 `DefaultNormalIndexNew()` 创建普通索引
6. **创建复合索引**：使用 `DefaultNormalIndexNew()` 创建复合索引
7. **创建全文索引**：使用 `DefaultFullTextIndexNew()` 创建全文索引
8. **插入数据**：使用 `Insert()` 插入多条商品数据
9. **主键搜索**：使用主键进行精确查询
10. **普通索引搜索**：使用普通索引进行字段查询
11. **复合索引搜索**：使用复合索引进行多字段查询
12. **全文索引搜索**：使用全文索引进行关键词搜索
13. **查看索引缓存统计**：通过 `GetIndexMatchCacheStats()` 查看缓存性能

## 4.2 组合主键（Composite Primary Key）

组合主键使用多个字段来唯一标识一条记录。在工业物联网场景中，常常需要用设备 ID + 时间戳的组合来唯一标识一条传感器读数。

**示例代码 1**（工业物联网场景）：

```go
// 创建设备读数表
table, err := TableNew("edgex_readings")
if err != nil {
    t.Fatalf("TableNew 失败: %v", err)
}

// 设置字段
fields := map[string]any{
    "id":         "",
    "deviceName": "",  // 设备名称
    "reading":    "",  // 读数类型
    "value":      0.0,
    "valueType":  "",
    "baseType":   "",
    "timestamp":  int64(0),  // 时间戳
    "metadata":   "",
}
table.SetFields(fields)

// 创建组合主键：设备名称 + 时间戳
PrimaryKeys, err := DefaultPrimaryKeyNew("pk")
if err != nil {
    t.Fatalf("DefaultPrimaryKeyNew 失败: %v", err)
}
PrimaryKeys.AddFields("deviceName", "timestamp")  // 组合主键
table.CreateIndex(PrimaryKeys)
```

**组合主键的优势**：

1. **业务语义清晰**：组合主键直接反映业务逻辑（如：设备 + 时间）
2. **无需自增ID**：可以直接用业务字段作为主键
3. **查询高效**：可以使用组合主键的前缀进行范围查询
4. **唯一约束**：确保多个字段的组合是唯一的
