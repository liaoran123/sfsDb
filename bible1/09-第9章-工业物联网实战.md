```go
package main

import (
	"encoding/json"
	"fmt"
	"math/rand"
	"os"
	"time"

	"github.com/liaoran123/sfsDb/engine"
	"github.com/liaoran123/sfsDb/storage"

type SensorData struct {
	ID        int64     `json:"id"`

	Value     float64     `json:"value"`
	Unit      string      `json:"unit"`
	Status    string      `json:"status"`
}

type Device struct {
	ID          string    `json:"id"`
	Name        string    `json:"name"`
	Type        string    `json:"type"`
	Location    string    `json:"location"`
	Status      string    `json:"status"`
	LastSeen    time.Time `json:"last_seen"`
}

type Alert struct {
	ID        int64     `json:"id"`
	SensorID  string    `json:"sensor_id"`
	Timestamp time.Time `json:"timestamp"`
	Type      string    `json:"type"`
	Message   string    `json:"message"`
	Level     string    `json:"level"`
	Resolved  bool      `json:"resolved"`

func main() {

	fmt.Println()

	dbPath := "./iot_edge_db"
	cleanup(dbPath)
	defer cleanup(dbPath)

	fmt.Println("1. 初始化加密数据库")
	masterKey := make([]byte, 32)
	for i := range masterKey {
		masterKey[i] = byte(i)
	}

	encryptConfig := &amp;storage.EncryptionConfig{
		Enabled:   true,
		Algorithm: "AES-256-GCM",
		MasterKey: masterKey,
	}

	_, err := storage.GetDBManager().OpenDB(dbPath, nil, encryptConfig)
	if err != nil {
		fmt.Printf("数据库打开失败: %v\n", err)
		return
	}
	defer storage.GetDBManager().CloseDB()
	fmt.Println("   ✓ 加密数据库初始化成功")
	fmt.Println()

	fmt.Println("2. 创建设备信息表")
	deviceTable, err := engine.TableNew("devices")
	if err != nil {
		fmt.Printf("创建设备表失败: %v\n", err)

	devicePK.AddFields("id")
	deviceTable.CreateIndex(devicePK)

	statusIndex, _ := engine.DefaultNormalIndexNew("idx_device_status")
	statusIndex.AddFields("status")
	deviceTable.CreateIndex(statusIndex)
	fmt.Println("   ✓ 设备信息表创建成功")
	fmt.Println()

	fmt.Println("3. 创建传感器数据表")
	sensorTable, err := engine.TableNew("sensor_data")
	if err != nil {
		fmt.Printf("创建传感器数据表失败: %v\n", err)
		return
	}

	sensorFields := map[string]any{
		"id":         int64(0),
		"sensor_id":   "",
		"timestamp":   time.Time{},
		"value":       float64(0),
		"unit":        "",
		"status":      "",
	}
	sensorTable.SetFields(sensorFields)

	sensorPK, _ := engine.DefaultPrimaryKeyNew("pk_sensor_id")
	sensorPK.AddFields("id")
	sensorTable.CreateIndex(sensorPK)

	sensorIDIndex, _ := engine.DefaultNormalIndexNew("idx_sensor_id")
	sensorIDIndex.AddFields("sensor_id")
	sensorTable.CreateIndex(sensorIDIndex)

	timestampIndex, _ := engine.DefaultNormalIndexNew("idx_sensor_timestamp")
	timestampIndex.AddFields("timestamp")
	sensorTable.CreateIndex(timestampIndex)
	fmt.Println("   ✓ 传感器数据表创建成功")
	fmt.Println()

	fmt.Println("4. 创建告警信息表")
	alertTable, err := engine.TableNew("alerts")

		fmt.Printf("创建告警表失败: %v\n", err)

	alertFields := map[string]any{
		"id":         int64(0),
		"sensor_id":   "",
		"timestamp":   time.Time{},
		"type":        "",

		"level":       "",
		"resolved":    false,

	alertTable.SetFields(alertFields)

	alertPK, _ := engine.DefaultPrimaryKeyNew("pk_alert_id")
	alertPK.AddFields("id")
	alertTable.CreateIndex(alertPK)

	levelIndex, _ := engine.DefaultNormalIndexNew("idx_alert_level")
	levelIndex.AddFields("level")
	alertTable.CreateIndex(levelIndex)

	resolvedIndex, _ := engine.DefaultNormalIndexNew("idx_alert_resolved")
	resolvedIndex.AddFields("resolved")
	alertTable.CreateIndex(resolvedIndex)
	fmt.Println("   ✓ 告警信息表创建成功")
	fmt.Println()

	fmt.Println("5. 注册边缘设备")
	devices := []Device{

		{

			LastSeen:    time.Now().Add(-2 * time.Hour),

	for _, device := range devices {

			"name":        device.Name,

	fmt.Println("   ✓ 成功注册", len(devices), "个边缘设备")
	fmt.Println()

	fmt.Println("6. 生成模拟传感器数据")
	now := time.Now()
	sensorDataCount := 0
	alertCount := 0

	for i := 0; i &lt; 100; i++ {
		sensorID := fmt.Sprintf("DEV-00%d", (i%3)+1)
		timestamp := now.Add(-time.Duration(99-i) * time.Minute)
		var value float64
		var unit string
		var status string

		switch i % 3 {
		case 0:
			value = 20 + rand.Float64()*10
			unit = "°C"
		case 1:
			value = 100 + rand.Float64()*50

				"sensor_id":   sensorID,
				"timestamp":   timestamp,
				"type":        "threshold_exceeded",
				"message":     fmt.Sprintf("温度超过阈值: %.1f°C", value),
				"level":       "warning",
				"resolved":    false,
			}
			alertTable.Insert(&amp;alertMap)
		}

		sensorDataCount++
		sensorMap := map[string]any{
			"id":         int64(sensorDataCount),
			"sensor_id":   sensorID,
			"timestamp":   timestamp,
			"value":       value,
			"unit":        unit,
			"status":      status,
		}
		sensorTable.Insert(&amp;sensorMap)
	}
	fmt.Printf("   ✓ 成功生成 %d 条传感器数据记录\n", sensorDataCount)
	fmt.Printf("   ✓ 成功生成 %d 条告警记录\n", alertCount)
	fmt.Println()

	fmt.Println("7. 查询在线设备")
	onlineSearch := map[string]any{"status": "online"}
	dataIter, _ := deviceTable.Search(&amp;onlineSearch)
	defer dataIter.Release()
	onlineRecords := dataIter.GetRecords(true)
	defer onlineRecords.Release()
	fmt.Println("   在线设备列表:")
	for _, d := range onlineRecords {
		fmt.Printf("   - %s (%s): %s\n", d["name"], d["id"], d["location"])
	}
	fmt.Println()

	fmt.Println("8. 查询最近 1 小时的传感器数据")
	oneHourAgo := now.Add(-1 * time.Hour)
	recentData := make([]map[string]any, 0)

	allSearch := map[string]any{"id": nil}
	allIter, _ := sensorTable.Search(&amp;allSearch)
	defer allIter.Release()
	allRecords := allIter.GetRecords(true)
	defer allRecords.Release()

	for _, record := range allRecords {
		ts, ok := record["timestamp"].(time.Time)
		if ok &amp;&amp; ts.After(oneHourAgo) {
			recentData = append(recentData, record)
		}
	}

	fmt.Printf("   最近 1 小时数据点数: %d\n", len(recentData))
	for _, data := range recentData[:5] {
		fmt.Printf("   - %s: %.1f %s\n", data["sensor_id"], data["value"], data["unit"])
	}
	if len(recentData) &gt; 5 {
		fmt.Println("   ...")
	}
	fmt.Println()

	fmt.Println("9. 查询未处理的告警")
	unresolvedSearch := map[string]any{"resolved": false}
	alertIter, _ := alertTable.Search(&amp;unresolvedSearch)
	defer alertIter.Release()
	alertRecords := alertIter.GetRecords(true)
	defer alertRecords.Release()
	fmt.Printf("   未处理告警数量: %d\n", len(alertRecords))
	for _, alert := range alertRecords {
		fmt.Printf("   - [%s] %s: %s\n", alert["level"], alert["sensor_id"], alert["message"])
	}
	fmt.Println()

	fmt.Println("10. 使用时序数据聚合示例")
	recordsForAgg := make([]map[string]any, 0)
	for _, record := range allRecords {
		if record["sensor_id"] == "DEV-001" {
			recordsForAgg = append(recordsForAgg, record)
		}
	}

	if len(recordsForAgg) &gt; 0 {
		startTime := now.Add(-2 * time.Hour)
		endTime := now
		window := sfstime.NewTumblingWindow(startTime, endTime, 30*time.Minute)
		results, _ := sfstime.AggregateByWindow(recordsForAgg, "timestamp", "value", window, "avg")
		fmt.Println("   DEV-001 温度每30分钟平均值:")
		for _, res := range results {
			fmt.Printf("   %s ~ %s: %.1f°C\n", res.WindowStart.Format("15:04"), res.WindowEnd.Format("15:04"), res.Value)
		}
	}
	fmt.Println()

	fmt.Println("11. 导出数据为 JSON")
	exportData := map[string]any{
		"devices":      onlineRecords,
		"recent_data": recentData,
		"alerts":      alertRecords,
		"export_time": time.Now(),
	}

	jsonData, _ := json.MarshalIndent(exportData, "", "  ")
	fmt.Println("   JSON 导出数据预览:")
	fmt.Println(string(jsonData[:200]))
	fmt.Println("   ...")
	fmt.Println()

	fmt.Println("12. 查看索引缓存统计")
	stats := engine.GetIndexMatchCacheStats()
	fmt.Printf("   索引缓存 - 大小: %d, 访问: %d, 命中: %d, 命中率: %.2f%%\n",
		stats.Size, stats.Accesses, stats.Hits, stats.HitRate*100)
	fmt.Println()

	fmt.Println("=== 工业物联网实战项目完成！===")
}
```

## 9.3 项目需求分析

在工业物联网（IIoT）场景中，边缘设备需要满足以下需求：

### 9.3.1 功能需求

1. **设备管理**：注册、管理和监控边缘设备
2. **数据采集**：实时采集传感器数据
3. **数据存储**：安全可靠地存储大量时序数据
4. **告警管理**：处理和管理设备告警信息
5. **数据查询**：灵活高效地查询历史数据
6. **数据分析**：对时序数据进行聚合和分析
7. **数据导出**：支持数据导出和集成

### 9.3.2 性能需求

1. **低延迟**：毫秒级数据读写响应
2. **高并发**：支持多传感器同时写入
3. **高可靠**：数据不丢失，支持故障恢复
4. **低功耗**：适合资源受限的边缘设备
5. **可扩展**：支持数据量增长

### 9.3.3 安全需求

1. **数据加密**：敏感数据加密存储
2. **访问控制**：限制数据访问权限
3. **数据完整性**：确保数据不被篡改
4. **审计日志**：记录数据操作日志

```go
// 传感器数据结构体
type SensorData struct {
    ID        int64     `json:"id"`
    SensorID  string      `json:"sensor_id"`
    Timestamp time.Time   `json:"timestamp"`
    Value     float64     `json:"value"`
    Unit      string      `json:"unit"`
    Status    string      `json:"status"`
}

// 写入传感器数据
func InsertSensorData(table *engine.Table, data SensorData) error {
    dataMap := map[string]any{
        "id":         data.ID,
        "sensor_id":   data.SensorID,
        "timestamp":   data.Timestamp,
        "value":       data.Value,
        "unit":        data.Unit,
        "status":      data.Status,
    }
    _, err := table.Insert(&amp;dataMap)
    return err
}

// 查询时间范围内的传感器数据
func GetSensorDataByTimeRange(table *engine.Table, sensorID string, startTime, endTime time.Time) ([]map[string]any, error) {
    allSearch := map[string]any{"id": nil}
    dataIter, err := table.Search(&amp;allSearch)
    if err != nil {
        return nil, err
    }
    defer dataIter.Release()
    
    records := dataIter.GetRecords(true)
    defer records.Release()
    
    var result []map[string]any
    for _, record := range records {
        ts, ok := record["timestamp"].(time.Time)
        if !ok {
            continue
        }
        sid, ok := record["sensor_id"].(string)
        if !ok {
            continue
        }
        if sid == sensorID &amp;&amp; ts.After(startTime) &amp;&amp; ts.Before(endTime) {
            result = append(result, record)
        }
    }
    return result, nil
}
```

```go
// 按时间窗口聚合传感器数据
func AggregateSensorData(table *engine.Table, sensorID string, window sfstime.TimeWindow, aggregationType string) ([]sfstime.WindowAggregation, error) {
    // 获取该传感器的所有数据
    allSearch := map[string]any{"id": nil}
    dataIter, err := table.Search(&amp;allSearch)
    if err != nil {
        return nil, err
    }
    defer dataIter.Release()
    
    records := dataIter.GetRecords(true)
    defer records.Release()
    
    // 筛选该传感器的数据
    var sensorRecords []map[string]any
    for _, record := range records {
        sid, ok := record["sensor_id"].(string)
        if ok &amp;&amp; sid == sensorID {
            sensorRecords = append(sensorRecords, record)
        }
    }
    
    // 进行聚合
    return sfstime.AggregateByWindow(sensorRecords, "timestamp", "value", window, aggregationType)
}
```
