## 9.1 完整的可运行示例

让我们通过一个完整的示例来演示 sfsDb 的匹配器功能：

```go
package main

import (
	"fmt"
	"os"

	"github.com/liaoran123/sfsDb/engine"
	"github.com/liaoran123/sfsDb/match"
	"github.com/liaoran123/sfsDb/storage"
)

func main() {
	fmt.Println("=== sfsDb 数据库实战 - 第 9 章：匹配器示例 ===")
	fmt.Println()

	dbPath := "./bible_examples_chapter09_db"
	os.RemoveAll(dbPath)
	defer os.RemoveAll(dbPath)

	fmt.Println("1. 初始化数据库")
	_, err := storage.GetDBManager().OpenDB(dbPath)
	if err != nil {
		fmt.Printf("数据库打开失败: %v\n", err)
		return
	}
	defer storage.GetDBManager().CloseDB()
	fmt.Println("   ✓ 数据库打开成功")
	fmt.Println()

	fmt.Println("2. 创建用户表")
	userTable, err := engine.TableNew("users")
	if err != nil {
		fmt.Printf("创建表失败: %v\n", err)
		return
	}

	fields := map[string]any{
		"id":     0,
		"name":   "",
		"age":    0,
		"email":  "",
		"score":  0.0,
		"active": false,
	}
	userTable.SetFields(fields)

	pk, _ := engine.DefaultPrimaryKeyNew("pk_id")
	pk.AddFields("id")
	userTable.CreateIndex(pk)
	fmt.Println("   ✓ 用户表创建成功")
	fmt.Println()

	fmt.Println("3. 插入测试数据")
	users := []map[string]any{
		{"id": 1, "name": "Alice", "age": 25, "email": "alice@example.com", "score": 85.5, "active": true},
		{"id": 2, "name": "Bob", "age": 30, "email": "bob@example.com", "score": 92.0, "active": true},
		{"id": 3, "name": "Charlie", "age": 22, "email": "charlie@test.com", "score": 78.5, "active": false},
		{"id": 4, "name": "David", "age": 35, "email": "david@example.com", "score": 95.0, "active": true},
		{"id": 5, "name": "Eve", "age": 28, "email": "eve@test.com", "score": 88.0, "active": false},
	}

	for _, u := range users {
		_, err = userTable.Insert(&u)
		if err != nil {
			fmt.Printf("插入数据失败: %v\n", err)
			return
		}
	}
	fmt.Println("   ✓ 成功插入", len(users), "条用户数据")
	fmt.Println()

	fmt.Println("4. 使用 FieldComparison 匹配器")
	fmt.Println("   查找年龄大于25岁的用户:")
	ageMatch := match.NewGreaterThanMatch("age", 25)
	searchAll := map[string]any{"id": nil}
	iter1, _ := userTable.Search(&searchAll)
	defer iter1.Release()
	iter1.SetMatch(ageMatch)
	records1 := iter1.GetRecords(true)
	defer records1.Release()
	for _, r := range records1 {
		fmt.Printf("   - %s (年龄: %d)\n", r["name"], r["age"])
	}
	fmt.Println()

	fmt.Println("5. 使用 Like 匹配器")
	fmt.Println("   查找邮箱以 @example.com 结尾的用户:")
	emailMatch := match.NewLikeMatch("email", "%@example.com")
	iter2, _ := userTable.Search(&searchAll)
	defer iter2.Release()
	iter2.SetMatch(emailMatch)
	records2 := iter2.GetRecords(true)
	defer records2.Release()
	for _, r := range records2 {
		fmt.Printf("   - %s: %s\n", r["name"], r["email"])
	}
	fmt.Println()

	fmt.Println("6. 使用 Contains 匹配器")
	fmt.Println("   查找名字包含 'a' 的用户（不区分大小写）:")
	nameMatch := match.NewContainsMatch("name", "a")
	iter3, _ := userTable.Search(&searchAll)
	defer iter3.Release()
	iter3.SetMatch(nameMatch)
	records3 := iter3.GetRecords(true)
	defer records3.Release()
	for _, r := range records3 {
		fmt.Printf("   - %s\n", r["name"])
	}
	fmt.Println()

	fmt.Println("7. 使用 AND 匹配器（IN 查询）")
	fmt.Println("   查找ID在 [1, 3, 5] 中的用户:")
	inData := map[any]bool{1: true, 3: true, 5: true}
	andMatch := match.NewAND([]string{"id"}, inData, true)
	iter4, _ := userTable.Search(&searchAll)
	defer iter4.Release()
	iter4.SetMatch(andMatch)
	records4 := iter4.GetRecords(true)
	defer records4.Release()
	for _, r := range records4 {
		fmt.Printf("   - %s (ID: %d)\n", r["name"], r["id"])
	}
	fmt.Println()

	fmt.Println("8. 使用自定义 FieldMatch")
	fmt.Println("   查找分数在80-90之间的用户:")
	customMatch := match.NewFieldMatch("score", func(val any) bool {
		if score, ok := val.(float64); ok {
			return score >= 80 && score <= 90
		}
		return false
	})
	iter5, _ := userTable.Search(&searchAll)
	defer iter5.Release()
	iter5.SetMatch(customMatch)
	records5 := iter5.GetRecords(true)
	defer records5.Release()
	for _, r := range records5 {
		fmt.Printf("   - %s (分数: %.1f)\n", r["name"], r["score"])
	}
	fmt.Println()

	fmt.Println("9. 组合多个匹配器")
	fmt.Println("   查找活跃用户且年龄小于30岁:")
	activeMatch := match.NewEqualMatch("active", true)
	youngMatch := match.NewLessThanMatch("age", 30)
	iter6, _ := userTable.Search(&searchAll)
	defer iter6.Release()
	iter6.SetMatch(activeMatch, youngMatch)
	records6 := iter6.GetRecords(true)
	defer records6.Release()
	for _, r := range records6 {
		fmt.Printf("   - %s (年龄: %d, 活跃: %v)\n", r["name"], r["age"], r["active"])
	}
	fmt.Println()

	fmt.Println("=== 匹配器示例完成！===")
}
```

## 9.2 Match 接口概述

match 包提供了强大的匹配功能，用于在查询时进行复杂的数据过滤。

**代码来源**: `match/match.go:7-11`

```go
// 根据TableIter回传的key,value所有能得到的field的值，进行所需匹配
type Match interface {
	//fields *map[string]any接收迭代器传回的值进行匹配
	Match(fields *map[string]any) bool
}
```

### 匹配器类型

match 包提供了多种匹配器类型：

1. **AND 匹配器**：用于 IN/NOT IN 查询
2. **FieldComparison**：字段比较匹配器（支持 >, <, =, !=, >=, <= 等）
3. **FieldMatch**：自定义函数匹配器

## 9.3 AND 匹配器（IN/NOT IN 查询）

AND 匹配器用于实现 SQL 中的 IN 和 NOT IN 查询功能。

**代码来源**: `match/match.go:19-52`

```go
type AND struct {
	//需要匹配的字段名，与fields *map[string]any中的key对应
	fields []string
	//data 是由(t *TableIter) Map(fields ...string) (data map[any]bool)生成
	//也可以自定义，比如sql语句中 field in (1,2,3)，则data=map[any]bool{1:true,2:true,3:true}
	data map[any]bool
	//rule是判断，true为对应sql语句的 IN 或 AND ，false为NOT IN 或 OR。
	rule bool
}

func NewAND(fields []string, data map[any]bool, rule ...bool) *AND {
	if len(rule) == 0 {
		rule = append(rule, true)
	}
	return &AND{
		fields: fields,
		data:   data,
		rule:   rule[0],
	}
}
```

### IN 查询示例

```go
// 查找 ID 在 [1, 3, 5] 中的用户
inData := map[any]bool{1: true, 3: true, 5: true}
andMatch := match.NewAND([]string{"id"}, inData, true)

iter, _ := table.Search(&searchAll)
defer iter.Release()
iter.SetMatch(andMatch)
records := iter.GetRecords(true)
defer records.Release()
```

### NOT IN 查询示例

```go
// 查找 ID 不在 [2, 4] 中的用户
notInData := map[any]bool{2: true, 4: true}
notMatch := match.NewAND([]string{"id"}, notInData, false)

iter, _ := table.Search(&searchAll)
defer iter.Release()
iter.SetMatch(notMatch)
records := iter.GetRecords(true)
defer records.Release()
```

### 复合字段匹配

```go
// 复合主键匹配
compositeData := map[any]bool{
	"device1_1234567890": true,
	"device2_1234567891": true,
}
compositeMatch := match.NewAND([]string{"deviceName", "timestamp"}, compositeData, true)
```

## 9.4 FieldComparison 匹配器

FieldComparison 提供了丰富的字段比较操作符。

**代码来源**: `match/FieldComparison.go:3-20`

```go
// 使用 FieldComparison.Match 进行比较操作
// 主要是用于主键迭代器对于无索引字段的匹配。
type FieldComparison struct {
	fieldName   string             // 要比较的字段名
	op          ComparisonOperator // 比较操作符
	value       any                // 比较值
	innerValues ValueComparer      // 内部使用的 ValueComparer 实例
}

// NewFieldComparison 创建一个新的 FieldComparison 实例
func NewFieldComparison(fieldName string, op ComparisonOperator, value any) *FieldComparison {
	return &FieldComparison{
		fieldName:   fieldName,
		op:          op,
		value:       value,
		innerValues: NewValueComparer(value),
	}
}
```

### 比较操作符

**代码来源**: `match/ComparisonOperator.go:1-28`

```go
// ComparisonOperator 定义比较操作符枚举
type ComparisonOperator int

// 比较操作符枚举值
const (
	// Equal 等于 (=)
	Equal ComparisonOperator = iota
	// NotEqual 不等于 (!=)
	NotEqual
	// GreaterThan 大于 (>)
	GreaterThan
	// GreaterThanOrEqual 大于等于 (>=)
	GreaterThanOrEqual
	// LessThan 小于 (<)
	LessThan
	// LessThanOrEqual 小于等于 (<=)
	LessThanOrEqual
	// Like 类似于SQL LIKE操作
	Like
	// Prefix 前缀匹配
	Prefix
	// Suffix 后缀匹配
	Suffix
	// Contains 包含匹配
	Contains
)
```

### LIKE 模式匹配

**代码来源**: `match/Values.go:93-108`

```go
// Like 检查当前值是否匹配类似SQL LIKE的模式（仅适用于字符串类型）
// 支持的通配符：
//
//	% - 匹配任意长度的字符串（包括空字符串）
//	_ - 匹配单个字符
func (m ValueComparer) Like(pattern string) bool {
	if str, ok := m.value.(string); ok {
		// 使用正则表达式处理所有LIKE模式
		// 替换所有%为.*，然后使用正则表达式匹配
		// 注意：这是一个简单实现，没有处理转义字符
		regexPattern := strings.ReplaceAll(pattern, "%", ".*")
		matched, _ := regexp.MatchString(regexPattern, str)
		return matched
	}
	return false
}
```

## 9.6 FieldMatch 自定义匹配器

FieldMatch 允许使用自定义函数进行匹配。

**代码来源**: `match/FieldComparison.go:40-69`

```go
// FieldMatch 是一个更简单的匹配器，用于直接比较字段值
// 它实现了 Match 接口，内部使用 ValueComparer 进行比较
type FieldMatch struct {
	fieldName string         // 要比较的字段名
	matchFunc func(any) bool // 匹配函数
}

// NewFieldMatch 创建一个新的 FieldMatch 实例
func NewFieldMatch(fieldName string, matchFunc func(any) bool) *FieldMatch {
	return &FieldMatch{
		fieldName: fieldName,
		matchFunc: matchFunc,
	}
}

// Match 实现了 Match 接口，使用自定义的匹配函数
func (fm *FieldMatch) Match(fields *map[string]any) bool {
	if fields == nil {
		return false
	}

	// 获取要比较的字段值
	fieldValue, exists := (*fields)[fm.fieldName]
	if !exists {
		return false
	}

	// 使用自定义的匹配函数进行比较
	return fm.matchFunc(fieldValue)
}
```

### 自定义匹配示例

```go
// 自定义范围匹配
rangeMatch := match.NewFieldMatch("score", func(val any) bool {
	if score, ok := val.(float64); ok {
		return score >= 80 && score <= 90
	}
	return false
})

// 自定义字符串长度匹配
lengthMatch := match.NewFieldMatch("name", func(val any) bool {
	if name, ok := val.(string); ok {
		return len(name) > 3
	}
	return false
})

// 自定义正则匹配
regexMatch := match.NewFieldMatch("email", func(val any) bool {
	if email, ok := val.(string); ok {
		matched, _ := regexp.MatchString(`^[a-zA-Z0-9]+@[a-zA-Z0-9]+\.[a-zA-Z]+$`, email)
		return matched
	}
	return false
})
```

### 匹配器的工作原理

**代码来源**: `engine/tableiter.go:234-244`

```go
/*
// 检测记录是否符合Match条件
// 如果符合，返回true
// 如果不符合，返回false
常见sql场景，f in (1,2,3) 或  and 等操作
*/
func (t *TableIter) Match(rd *map[string]any, match []match.Match) bool {
	if len(match) == 0 {
		return true //不需要匹配
	}
	for _, m := range match {
		if !m.Match(rd) {
			return false
		}
	}
	return true
}
```

### 使用多个匹配器

```go

	match.NewEqualMatch("active", true),
	match.NewGreaterThanMatch("age", 25),
	match.NewLessThanMatch("age", 30)
)
```
