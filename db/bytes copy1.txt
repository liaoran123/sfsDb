package db

import (
	"bytes"
	"encoding/binary"
	"math"
	"time"
)

type Bytes []byte

func (b Bytes) Bool() bool {
	return b[0] == 1
}

func (b Bytes) Int() int64 {
	bytesBuffer := bytes.NewBuffer(b)
	var x int64
	if IsLitEndian {
		binary.Read(bytesBuffer, binary.LittleEndian, &x)
	} else {
		binary.Read(bytesBuffer, binary.BigEndian, &x)
	}
	return x
}

func (b Bytes) Float() float64 {
	var bits uint64
	if IsLitEndian {
		bits = binary.LittleEndian.Uint64(b)
	} else {
		bits = binary.BigEndian.Uint64(b)
	}
	return math.Float64frombits(bits)
}

func (b Bytes) String() string {
	return string(b)
}

// 如果时间格式错误，返回2006-01-02 15:04:05
func (b Bytes) Time() time.Time {
	t, err := time.Parse("2006-01-02 15:04:05", string(b))
	if err != nil {
		t, err = time.Parse("2006-01-02 15:04:05", "2006-01-02 15:04:05")
	}
	return t
}

// 格式化数据
// 用左右括号包括数据，并且进行转义
// 数据格式：字段id(数据)。如：1(abcde)2(12345)
func (b Bytes) Format(id byte) []byte {
	//b = bytes.Join([][]byte{[]byte{id, '('}, b, []byte{')'}}, []byte{})
	b = bytes.Join([][]byte{{id, LEFT_SPLIT[0]}, b.Escape(), {RIGHT_SPLIT[0]}}, []byte{})
	return b
}

// 反格式化数据
// 去掉id和左右括号。1(abcde) => abcde。并且反转义，将重复的括号还原为单个括号
func (b Bytes) UnFormat() []byte {
	r := b[2 : len(b)-1]
	return r.UnEscape()
}

// 对数据进行转义，左右括号进行转义，遇到括号时,重复写入两次,作为转义
func (b Bytes) Escape() []byte {
	/* 效率低
	b = bytes.ReplaceAll(b, []byte{LEFT_SPLIT[0]}, []byte{LEFT_SPLIT[0], LEFT_SPLIT[0]})
	b = bytes.ReplaceAll(b, []byte{RIGHT_SPLIT[0]}, []byte{RIGHT_SPLIT[0], RIGHT_SPLIT[0]})
	return b
	*/
	blen := len(b)

	var buffer bytes.Buffer
	leftSPLIT := LEFT_SPLIT[0]
	rightSPLIT := RIGHT_SPLIT[0]
	for i := range blen {
		c := b[i]
		switch c {
		case leftSPLIT:
			buffer.Write([]byte{leftSPLIT, leftSPLIT}) // 遇到分隔符时，重复写入两次
		case rightSPLIT:
			buffer.Write([]byte{rightSPLIT, rightSPLIT}) // 遇到分隔符时，重复写入两次

		default:
			buffer.WriteByte(c)
		}
	}
	return buffer.Bytes()
}

// 反转义
func (b Bytes) UnEscape() []byte {
	blen := len(b)
	var buffer bytes.Buffer
	left := LEFT_SPLIT[0]
	right := RIGHT_SPLIT[0]
	for i := 0; i < len(b)-1; i++ {
		c := b[i]
		if i+1 < blen {
			switch c {
			case left:
				if b[i+1] == left { //连续2个分隔符，表示数据中有分隔符
					buffer.WriteByte(left) //只需添加1个分隔符到数据中
					i++                    //跳过下一个分隔符
					continue
				}
			case right:
				if b[i+1] == right { //连续2个分隔符，表示数据中有分隔符
					buffer.WriteByte(right) //只需添加1个分隔符到数据中
					i++                     //跳过下一个分隔符
					continue
				}
			}
		}
		buffer.WriteByte(c)
	}
	return buffer.Bytes()
}

// 合并多个
func (b Bytes) Jion(ib ...[]byte) []byte {
	for _, v := range ib {
		b = bytes.Join([][]byte{b, v}, []byte{})
	}
	return b //原本的 b值，并不能被改变，所有需要返回一个新的值
}

					s.WriteByte(rightSPLIT) //只需添加1个分隔符到数据中
					i++                     //跳过下一个分隔符
					continue
				}
				/*
					case leftSPLIT:

		}
		if c == rightSPLIT {

		} else {

		}

		rs = append(rs, s.Bytes())
	}
	return rs
}
